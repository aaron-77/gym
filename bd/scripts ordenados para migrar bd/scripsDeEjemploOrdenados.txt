////////////////////INICIA PK////////////////////////

$dir  = Join-Path $basedir "060_primary_keys"
New-Item -ItemType Directory -Force -Path $dir | Out-Null

$file = Join-Path $dir "060_primary_keys.sql"

$options = New-DbaScriptingOption
$options.DriPrimaryKey = $true
$options.PrimaryObject = $false
$options.ScriptSchema  = $true
$options.AppendToFile  = $true
$options.FileName      = $file

Remove-Item $file -ErrorAction Ignore

Get-DbaDbTable -SqlInstance $server -Database $db | ForEach-Object {
    $_.Indexes | Where-Object IsPrimaryKey | ForEach-Object {
        $_.Script($options)
    }
}

///////////////////FIN PK////////////////////////////

////////////////////INICIO UNIQUE CONSTRAINTS///////////////
$dir = Join-Path $basedir "070_unique_constraints"
New-Item -ItemType Directory -Force -Path $dir | Out-Null

$options = New-DbaScriptingOption
$options.DriUniqueKeys = $true
$options.PrimaryObject = $true
$options.ScriptSchema  = $true

$i = 1
Get-DbaDbTable -SqlInstance $server -Database $db | ForEach-Object {
    $_.Indexes |
    Where-Object { $_.IsUnique -and -not $_.IsPrimaryKey } |
    ForEach-Object {
        $options.FileName = Join-Path $dir ("{0:D3}_UQ_{1}.sql" -f $i, $_.Name)
        $_.Script($options)
        $i++
    }
}

///////////////////FIN UNIQUE CONSTRAINTS///////////////////

////////////////////////INICIO CHECK CONSTRAINTS///////////
$dir = Join-Path $basedir "090_check_constraints"
New-Item -ItemType Directory -Force -Path $dir | Out-Null

$options = New-DbaScriptingOption
$options.DriChecks     = $true
$options.PrimaryObject = $true

$i = 1
Get-DbaDbTable -SqlInstance $server -Database $db | ForEach-Object {
    $_.Checks | ForEach-Object {
        $options.FileName = Join-Path $dir ("{0:D3}_CK_{1}.sql" -f $i, $_.Name)
        $_.Script($options)
        $i++
    }
}


/////////////////////// FIN CHECK CONSTRAINTS//////////////

///////////////////////INICIO DEFAULT  CONSTRAINTS//////////

$dir = Join-Path $basedir "100_default_constraints"
New-Item -ItemType Directory -Force -Path $dir | Out-Null

$options = New-DbaScriptingOption
$options.DriDefaults   = $true
$options.PrimaryObject = $true

$i = 1
Get-DbaDbTable -SqlInstance $server -Database $db | ForEach-Object {
    $_.Columns |
    Where-Object { $_.DefaultConstraint } |
    ForEach-Object {
        $options.FileName = Join-Path $dir ("{0:D3}_DF_{1}.sql" -f $i, $_.DefaultConstraint.Name)
        $_.DefaultConstraint.Script($options)
        $i++
    }
}

///////////////////INICIO INDICES NO PK NO UNIQUE///////////

$dir = Join-Path $basedir "110_indexes"
New-Item -ItemType Directory -Force -Path $dir | Out-Null

$options = New-DbaScriptingOption
$options.Indexes        = $true
$options.PrimaryObject  = $true
$options.ClusteredIndexes    = $true
$options.NonClusteredIndexes = $true

$i = 1
Get-DbaDbTable -SqlInstance $server -Database $db | ForEach-Object {
    $_.Indexes |
    Where-Object {
        -not $_.IsPrimaryKey -and
        -not $_.IsUniqueConstraint
    } |
    ForEach-Object {
        $options.FileName = Join-Path $dir ("{0:D3}_IDX_{1}.sql" -f $i, $_.Name)
        $_.Script($options)
        $i++
    }
}
